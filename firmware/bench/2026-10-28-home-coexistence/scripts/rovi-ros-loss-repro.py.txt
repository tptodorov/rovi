"""Short observational ROS Echo check; fails if any of 300 replies are absent."""
import json
from pathlib import Path
import re
import subprocess
import sys

sys.path.insert(0,'/tmp/rovi-61')
import bench
ROOT=Path('/home/todor/tptodorov/rovi')
source=Path(sys.argv[1]) if len(sys.argv)>1 else ROOT/'apps/simcar/scenarios/ros_latency.py'
out=Path('/tmp/rovi-ros-loss-'+sys.argv[2])
out.mkdir(mode=0o700,exist_ok=False)
log=out/'car.log'
monitor=bench.monitor(str(log))
try:
    found,_=bench.wait(str(log),lambda e:e.get('ev')=='net' and e.get('mode')=='Home',25)
    assert found,'Home unavailable'
    result=subprocess.run(['docker','run','--rm','--net','host',
        '-v','/tmp/rovi-client-lan.json5:/c.json5:ro','-v',str(source)+':/probe.py:ro',
        '-e','ZENOH_SESSION_CONFIG_URI=/c.json5','-e','ROVI_N=300','-e','ROVI_HZ=20',
        'rovi-ros-sim','bash','-c','source /opt/ros/jazzy/setup.bash && python3 /probe.py'],capture_output=True,text=True,timeout=45)
    (out/'ros.txt').write_text(result.stdout+result.stderr)
    assert result.returncode==0,(result.returncode,result.stderr[-600:])
    print(result.stdout,flush=True)
    events=bench.carlog.events(log)
    assert not [e for e in events if e['ev'] in ('claimed','released','stop','error','dropped') or e.get('state')=='Armed' or any(e.get('wheels_pm',[]))]
    match=re.search(r'ros sent=(\d+) count=(\d+)',result.stdout)
    assert match,'missing measurement'
    sent,received=map(int,match.groups())
    (out/'result.json').write_text(json.dumps({'source':str(source),'sent':sent,'received':received,'lost':sent-received},indent=2)+'\n')
    assert sent==received==300,f'{sent-received}/{sent} ROS Echo replies missing'
finally:
    bench.stop(monitor)
