def audit():
 assert monitor.poll() is None,'serial monitor exited'
 events=carlog.events(raw/'car.log')
 assert sum(e.get('ev')=='boot' for e in events)==1,'unexpected reset or missing boot'
 assert not any(e.get('ev') in ('claimed','released','stop','error','dropped') for e in events),'unexpected ownership/safety/fault event'
 assert not any(e.get('state')=='Armed' or any(e.get('wheels_pm',[])) for e in events),'unexpected arming/nonzero command'
 assert not carlog.PANIC.search(carlog.read(raw/'car.log')),'panic'

def run_probe(name,seconds):
 global probe
 with (raw/f'{name}.txt').open('wb') as transcript:
  command=['rtk','proxy','python3','firmware/bench/wifi_soak.py','--target','192.168.4.1:7777','--interface',interface,'--seconds',str(seconds),'--output',str(raw/name)]
  if name=='soak':command=['rtk','proxy','bash','firmware/bench/ros_wifi_soak.sh',interface,str(seconds),str(raw/name)]
  env={**os.environ,'ROVI_TARGET':'192.168.4.1:7777','ROVI_CARLOG':str(raw/'car.log'),'ROVI_ROUTER_PORT':'7447'}
  probe=subprocess.Popen(command,env=env,stdout=transcript,stderr=subprocess.STDOUT,start_new_session=True)
  end=time.monotonic()+seconds+100
  last_audit=0
  while probe.poll() is None:
   if time.monotonic()-last_audit>.5:
    audit();last_audit=time.monotonic()
   assert time.monotonic()<end,'probe exceeded bounded observation/setup time'
   time.sleep(.1)
  audit()
  assert probe.returncode in (0,1),f'probe failed: {probe.returncode}'
  summary=json.loads((raw/name/'summary.json').read_text())
  assert summary['complete'] and summary['sent']>0,'observation incomplete/no samples'
  note(f'{name} finished: {summary}')
  return summary
